# beebots settings. Most people need NONE of this: the Setup page asks for your keys when you first open the site.
# Copy to .env (next to docker-compose.yml) only if you want to change something. NEVER commit .env.

# ---- Mode -------------------------------------------------------------------
# dry  = PAPER TRADING (the default): real market prices, simulated money, no exchange account needed
# demo = OKX demo trading with per-bee DEMO keys, fake money on OKX's side
# live = REAL MONEY. Needs DRY_RUN=false, MODE=live, per-bee LIVE keys AND LIVE_ACK below. Not financial advice.
MODE=dry
# Safety latch. true (the default, also when unset) forces paper trading whatever MODE says.
DRY_RUN=true
# Only for MODE=live: must be exactly I-ACCEPT-REAL-MONEY-RISK. You can lose everything you put in.
LIVE_ACK=

# ---- Setup ------------------------------------------------------------------
# The Setup page stays open this many minutes after the engine starts. First come, first served: once saved it closes
# for good. If nobody finishes in time, restart the engine container to open it again.
SETUP_WINDOW_MIN=120
# The owner password is picked on Setup and stored as a hash. Set this only if your Setup file has none (an install
# from before it existed, or keys set only here). At least 8 characters. Gates Join/Leave the Hive on the dashboard.
OWNER_PASSWORD=

# ---- Decisions: Gemini 4 Argon (or Jev) -------------------------------------
# Who picks each bee's next move. Setup asks for one key; anything set here wins over Setup.
#   GEMINI_API_KEY set           -> Gemini 4 Argon (wins if both keys are set)
#   only TYPESAFE_API_KEY set    -> Jev
#   neither                      -> a deterministic code chooser (paper and demo only; MODE=live refuses to start)
# Over the daily cap, or when the model is down or answers off the menu, the code chooser answers until it is back.
# It does not read a bee's free-text rules (coin limits and the risk layer still apply). Stops and caps are code.

# Gemini 4 Argon. Key from Google AI Studio (https://aistudio.google.com/apikey), in a project with billing on:
# the free tier cannot use Argon. README: "Run it with Gemini 4 Argon".
GEMINI_API_KEY=
GEMINI_MODEL=gemini-4-argon
GEMINI_TIMEOUT_MS=3000
# Daily Gemini spend cap in USD as the engine counts it, for all three bees together. Each call reserves its worst case
# before it is sent; a call that does not fit is not sent (the code chooser answers, and the dashboard says so).
GEMINI_DAILY_USD_CAP=3
# USD per million tokens, input and output (thinking counts as output). Both must be above 0. The defaults are Argon's
# price after the launch offer ($4 / $20). Until 30 Nov 2026 the launch price is $2 / $10 (announced by Google on
# 13 Oct 2026), so until then the counter prices tokens at twice the billed rate. Check Google's pricing page.
GEMINI_USD_PER_MTOK=4
GEMINI_USD_PER_MTOK_OUT=20
# At most one paid call per bee per this many ms, whatever TICK_MS is. In between, the bee keeps what it has.
GEMINI_MIN_INTERVAL_MS=60000
# Optional thinkingConfig.thinkingLevel (blank = the model's default).
GEMINI_THINKING_LEVEL=

# Jev (TypeSafe AI). Key from https://console.typesafe.ai/keys
TYPESAFE_API_KEY=
JEV_MODEL=jev-1.13.0
JEV_TIMEOUT_MS=2000
# Daily Jev spend threshold in USD (soft): no new Jev call once today's spend plus the calls in flight would pass it.
# Jev bills by its own token count, so calls already in flight can take a day somewhat over it. JEV_USD_PER_MTOK must be
# above 0.
JEV_DAILY_USD_CAP=2

# How often each bee is looked at (ms): stops, caps and the code chooser run every tick. Jev is asked every tick
# (10000 = 18 decisions/min for three bees, about $0.90/day, docs/COSTS.md); Gemini at most every GEMINI_MIN_INTERVAL_MS.
TICK_MS=10000
# Full market refresh (candles, indicators, funding, OI): every 60 s
DATA_REFRESH_MS=60000

# ---- OpenAI (optional) --------------------------------------------------------
# Setup asks for this too (required there). Used only on the Setup page: design each bee from a sentence, and paint it.
OPENAI_API_KEY=
OPENAI_TEXT_MODEL=gpt-5.4-nano
OPENAI_IMAGE_MODEL=gpt-image-2

# ---- Risk (every bee) ---------------------------------------------------------
